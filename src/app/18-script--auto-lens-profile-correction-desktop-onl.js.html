('+exif.lens+')':'')+'.'; status.style.color='var(--ok, #4caf50)'; return; } status.style.color=''; if(window.chromasmithLensOverride){ status.textContent='Not applied — no distortion calibration for "'+window.chromasmithLensOverride+'" in the bundled database (it may still be listed for vignette/CA-only data). Using the manual sliders below.'; return; } if(!exif.model&&!exif.lens){ status.textContent='This photo hasn’t finished loading its metadata yet — status will update automatically.'; return; } if(!exif.lens){ status.textContent='This photo’s RAW metadata has no lens model recorded — try the Manual lens picker above if you\'re using an adapted/manual lens.'; return; } status.textContent='Not applied — no bundled profile for "'+(exif.model||'this camera')+' · '+exif.lens+'" yet. Using the manual sliders below, or try the Manual lens picker above.'; } window.chromasmithLensStatusRefresh=fxLensStatusRefresh; function fxLensAutoToggled(){ const on=document.getElementById('tg-lens-auto').classList.contains('on'); window.chromasmithAutoLens=on; try{localStorage.setItem('chromasmithAutoLens',on?'1':'0');}catch(e){} if(!window.__TAURI__){fxLensStatusRefresh();return;} if(!on){ fxLensStatusRefresh(); if(typeof window.chromasmithReloadCurrentPhoto==='function')window.chromasmithReloadCurrentPhoto(); return; } const status=document.getElementById('lens-auto-status'); if(status)status.textContent='Reloading this photo to apply…'; const reloaded=(typeof window.chromasmithReloadCurrentPhoto==='function')&&window.chromasmithReloadCurrentPhoto(); if(!reloaded)fxLensStatusRefresh(); // no open photo to reload — show why nothing happened } if(typeof window!=='undefined')(()=>{ // Windows has no OS-level overlay scrollbars, so `.fx-scrolling` (CSS above) fakes one: // scroll events don't bubble, so this listens in the capture phase on the document. const fxScrollTimers=new WeakMap(); document.addEventListener('scroll',e=>{ const el=e.target===document?document.documentElement:e.target; if(!el||!el.classList)return; el.classList.add('fx-scrolling'); clearTimeout(fxScrollTimers.get(el)); fxScrollTimers.set(el,setTimeout(()=>el.classList.remove('fx-scrolling'),800)); },{capture:true,passive:true}); })(); if(typeof window!=='undefined')window.addEventListener('DOMContentLoaded',()=>{ const tg=document.getElementById('tg-lens-auto'); if(tg&&window.chromasmithAutoLens)tg.classList.add('on'); fxLensStatusRefresh(); fxLensManualInit(); });
