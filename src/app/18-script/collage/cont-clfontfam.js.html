ctx.fillStyle=fi.col; if(fi.deckle){ctx.beginPath();clDeckPts(R.w,R.h,i+1).forEach((q,k)=>k?ctx.lineTo(R.x+q[0],R.y+q[1]):ctx.moveTo(R.x+q[0],R.y+q[1]));ctx.closePath();} else clRoundRect(ctx,R.x,R.y,R.w,R.h,rad); ctx.fill();ctx.restore(); if(CL.frame==='print')matCap={x:R.x+fi.l,y:R.y+R.h-fi.b,h:fi.b,w:R.w-fi.l-fi.r}; r={x:R.x+fi.l,y:R.y+fi.t,w:R.w-fi.l-fi.r,h:R.h-fi.t-fi.b}; } ctx.save();if(fi&&!fi.round){ctx.beginPath();ctx.rect(r.x,r.y,r.w,r.h);}else clRoundRect(ctx,r.x,r.y,r.w,r.h,rad);ctx.clip(); const cover=Math.max(r.w/iw,r.h/ih)*c.scale,dw=iw*cover,dh=ih*cover; const mx=Math.max(0,(dw-r.w)/(2*r.w)),my=Math.max(0,(dh-r.h)/(2*r.h)); // keep the image covering the cell const ox=Math.min(mx,Math.max(-mx,c.ox)),oy=Math.min(my,Math.max(-my,c.oy)); ctx.drawImage(img,r.x+(r.w-dw)/2+ox*r.w,r.y+(r.h-dh)/2+oy*r.h,dw,dh); const cap=clCaption(i); if(cap&&matCap){ctx.restore();ctx.save();ctx.fillStyle=CL.capColor||'#222';ctx.font=`${tg.capPx}px ${fam}`;ctx.textBaseline='middle';ctx.textAlign='left'; ctx.fillText(cap,matCap.x+tg.capPx*0.6,matCap.y+matCap.h/2,matCap.w-tg.capPx*1.2);} else if(cap){if(!CL.capColor){const ch=tg.capPx*2.4,g=ctx.createLinearGradient(0,r.y+r.h-ch,0,r.y+r.h);g.addColorStop(0,'rgba(0,0,0,0)');g.addColorStop(1,'rgba(0,0,0,.55)'); ctx.fillStyle=g;ctx.fillRect(r.x,r.y+r.h-ch,r.w,ch);} ctx.fillStyle=CL.capColor||'#fff';ctx.font=`${tg.capPx}px ${fam}`;ctx.textBaseline='alphabetic';ctx.textAlign='left'; ctx.fillText(cap,r.x+tg.capPx*0.6,r.y+r.h-tg.capPx*0.55,r.w-tg.capPx*1.2);} ctx.restore(); }
