// ── Auto-enhance ─────────────────────────────────────────────────────────────── let _autoEnh=null; // values the last Auto-enhance applied; a second click while they still hold resets Light const _AE_IDS=['sl-adj-exp','sl-adj-bl','sl-adj-wh','sl-adj-temp','sl-adj-tint','sl-adj-hi','sl-adj-sh','sl-adj-con','sl-adj-vib']; // Pure analysis of an RGBA thumbnail -> Basic slider values. Works in linear light (so "exposure" is // real stops, not a gamma-space nudge), protects highlights before brightening, lifts shadows and // trims highlights by how much of the frame is actually crushed/blown, sets contrast from tonal // spread, and only touches white balance when enough near-neutral pixels make a gray-world estimate // trustworthy. Every move is damped and clamped: a good photo should barely change. function autoEnhanceParams(data){ const cl=(v,lo,hi)=>Math.max(lo,Math.min(hi,v)),n=data.length>>2; const s2l=c=>c<=0.04045?c/12.92:Math.pow((c+0.055)/1.055,2.4); const lin=new Float32Array(n),sl=new Float32Array(n); let logSum=0,sumR=0,sumG=0,sumB=0,neutral=0,chromaSum=0; for(let i=0;i0.12&&sl[i]<0.88&&ch<0.22){neutral++;sumR+=r;sumG+=g;sumB+=b;} } const sortedL=Float32Array.from(lin).sort(),sortedS=Float32Array.from(sl).sort(); const q=(a,p)=>a[Math.min(n-1,Math.floor(n*p))]; const geo=Math.exp(logSum/n),med=q(sortedL,0.5),p995=q(sortedL,0.995); // Key: blend of geometric mean and median, so a bright sky or dark void doesn't set exposure alone. const key=Math.max(1e-3,0.5*geo+0.5*med); let stops=Math.log2(0.16/key)*0.8; // 0.16 linear ~ sRGB 0.44, a typical well-exposed key stops=cl(stops,-1.5,2); // Never brighten into clipping: keep p99.5 under ~0.98 linear; the Highlights slider recovers the rest. if(stops>0&&p995>1e-3)stops=Math.min(stops,Math.max(0,Math.log2(0.98/p995)+0.5)); const gain=Math.pow(2,stops); let clipN=0,darkN=0; for(let i=0;i0.8||lin[i]>0.9)clipN++;if(v<0.004)darkN++;} const p1s=q(sortedS,0.01),p99s=q(sortedS,0.99),p5s=q(sortedS,0.05),p95s=q(sortedS,0.95); const exp=Math.round(cl(stops*20,-100,100)); const hi=Math.round(cl(-(clipN/n)*250,-60,0)); // 24% near-white pixels -> -60 const sh=Math.round(cl((darkN/n)*250,0,50)); // lift only when real shadow area exists const bl=Math.round(cl(-p1s*60,-15,0)); // gentle: sets black point, never crushes const wh=Math.round(cl((1-p99s)*60,0,15)); const con=Math.round(cl((0.62-(p95s-p5s))*70,-20,25)); const vib=Math.round(cl((0.14-chromaSum/n)*180,0,22)); let temp=0,tint=0; if(neutral/n>0.06){ const r=sumR/neutral,g=sumG/neutral,b=sumB/neutral; temp=Math.round(cl((b-r)/(r+b+1e-9)*100/0.22*0.6,-40,40)); tint=Math.round(cl((1-(r+b)/(2*g+1e-9))/0.14*100*0.6,-30,30)); } return{exp,bl,wh,temp,tint,hi,sh,con,vib}; } function autoEnhance(){ const src=fxWork||fxImg;if(!src){log('Load an image first','warn');return;} const aeBtn=document.querySelector('[onclick="autoEnhance()"]'); if(_autoEnh&&_AE_IDS.every((id,i)=>+document.getElementById(id)?.value===_autoEnh[i])){_autoEnh=null;aeBtn?.classList.remove('on');fxResetSection('adjust');return;} const sw=src.naturalWidth||src.width,sh0=src.naturalHeight||src.height; const mw=Math.min(sw,400),mh=Math.max(1,(mw*sh0/sw)|0); const tc=document.createElement('canvas');tc.width=mw;tc.height=mh; tc.getContext('2d').drawImage(src,0,0,mw,mh); const P=autoEnhanceParams(tc.getContext('2d').getImageData(0,0,mw,mh).data); const setSlider=(id,v)=>{const el=document.getElementById(id);if(el)el.value=v;}; setSlider('sl-adj-exp',P.exp);setSlider('sl-adj-bl',P.bl);setSlider('sl-adj-wh',P.wh); setSlider('sl-adj-temp',P.temp);setSlider('sl-adj-tint',P.tint); setSlider('sl-adj-hi',P.hi);setSlider('sl-adj-sh',P.sh);setSlider('sl-adj-con',P.con);setSlider('sl-adj-vib',P.vib); const tg=document.getElementById('tg-adjust');if(!tg.classList.contains('on'))toggleFX('adjust'); _autoEnh=[P.exp,P.bl,P.wh,P.temp,P.tint,P.hi,P.sh,P.con,P.vib];aeBtn?.classList.add('on'); fxUpdate(); log(`Auto-enhance: exp ${P.exp}, highlights ${P.hi}, shadows ${P.sh}, contrast ${P.con}, blacks ${P.bl}, whites ${P.wh}, vibrance ${P.vib}, temp ${P.temp}, tint ${P.tint}`,'ok'); }
