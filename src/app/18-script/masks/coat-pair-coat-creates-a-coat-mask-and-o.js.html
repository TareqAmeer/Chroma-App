m.coatDepth=v; m.exp=-0.6*(v/100); // up to -0.6 stops, darker m.sat=0.55*(v/100); // up to +0.55 saturation, more vivid red // ⚠️ Ease off the darkening on already-bright/overexposed fur — but ease off, don't exempt it. // First attempt (lumHi=0.72) protected brights almost completely (measured 97% less darkening // than midtones) — that traded one glaring mismatch for another: an almost-untouched bright // patch sitting next to a much-darker treated coat reads as its OWN harsh seam, just inverted. // Measured on a real backlit photo (geneva/__TM3938.jpg's sun-catching ear) at a few lumHi // values and picked the one where bright fur still visibly deepens, just less than midtones — // 0.80 measures 54% LESS darkening on the brightest fur (dropped 0.047 vs midtones' 0.071 at // the same strength), a difference of degree rather than an on/off boundary. mskD's lum-CEILING // gate (already wired into every mask's own selection weight, smoothstep-feathered not a hard // cutoff — see mskShapeWeight) is what does this, fading the mask's effect out gradually as a // pixel's own pre-existing luminance climbs. m.lumHi=v>0?0.80:1; const lb=document.getElementById('msk-coatdepth-val');if(lb)lb.textContent=Math.round(v); fxUpdate(); } function mskEyeBright(v){ const m=fxState.masks[mskSel];if(!m||m.origin!=='coatEye')return; m.eyeBright=v;
