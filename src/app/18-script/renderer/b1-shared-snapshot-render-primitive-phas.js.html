// ── B1: shared snapshot-render primitive (Phase B) ──────────────────────────── // Renders an ARBITRARY saved UI snapshot into a target 2D canvas without disturbing the live // editing state — used by split-compare, the history scrub rail, and (later) Library compare // panes, so they all share one code path instead of three near-duplicate render branches. // snap === null → render the CURRENT live state (no swap at all) // snap === 'orig' → the existing all-FX-off literal (geometry/crop is already baked into // fxWork by updateWork(), so this needs no UI-state swap either) // snap === {...} → a full getUISnapshot()-shaped object (history entry / style / pin / export) const FX_ORIG_PARAMS={lut:null,adjust:{enabled:false},grain:{enabled:false},halation:{enabled:false}, bloom:{enabled:false},debug:0,bordersEnabled:false,b1:{thickness:0},b2:{thickness:0}}; function renderSnapshotTo(dstCanvas,snap,pw,ph,opts){ opts=opts||{}; const blit=()=>{dstCanvas.width=pw;dstCanvas.height=ph;dstCanvas.getContext('2d').drawImage(FX.cv,0,0);}; if(snap==='orig'){ FX.render(FX_ORIG_PARAMS,pw,ph,{glowScale:fxGlowScale(),...opts}); blit();return; } if(snap===null){ FX.render(getFXParams(),pw,ph,{glowScale:fxGlowScale(),...opts}); blit();return; } // Full UI-snapshot swap: MUST be synchronous start-to-finish (no await/rAF) or the DOM // sliders would visibly flicker to the snapshot's values and race fxUpdate's debounce. const live=getUISnapshot(); _fxHistLocked=true; // reuse the existing guard (fxUndo/fxRedo/fxResetAll) — else applyUISnapshot // would push a history entry, fire chromasmithOnEdit (sidecar write!) and autosave try{ applyUISnapshot(snap); FX.render(getFXParams(),pw,ph,{glowScale:fxGlowScale(),...opts}); blit(); } finally { applyUISnapshot(live); _fxHistLocked=false; } }
